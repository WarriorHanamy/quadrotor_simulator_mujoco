# Standalone build for quadrotor_sim_core on Linux hosts without ROS
# Usage (direct): cmake -B build -S .
# Usage (via trampoline): include() from another CMakeLists.txt
#
# Paths use CMAKE_CURRENT_LIST_DIR so the file works from any -S directory.

cmake_minimum_required(VERSION 3.10)
project(quadrotor_sim_core_standalone)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Resolve project root relative to this CMakeLists file
get_filename_component(PROJ_ROOT "${CMAKE_CURRENT_LIST_DIR}" ABSOLUTE)

# MuJoCo library
add_library(mujoco SHARED IMPORTED)
set_target_properties(mujoco PROPERTIES
  IMPORTED_LOCATION ${PROJ_ROOT}/lib/libmujoco.so)

# Core simulator sources
add_executable(quadrotor_sim_core
  ${PROJ_ROOT}/src/core/main.cc
  ${PROJ_ROOT}/src/core/sim_core.cc
)

target_include_directories(quadrotor_sim_core PRIVATE
  ${PROJ_ROOT}/include
  ${PROJ_ROOT}/src/core
)

target_compile_options(quadrotor_sim_core PRIVATE -Wall -Wextra)

target_link_libraries(quadrotor_sim_core PRIVATE
  mujoco
  pthread
  rt
  ${CMAKE_DL_LIBS}
)

# GLFW adapter (optional, requires GLFW)
find_package(PkgConfig)
pkg_check_modules(GLFW glfw3)

if(GLFW_FOUND)
  add_executable(quadrotor_sim_glfw_adapter
    ${PROJ_ROOT}/src/glfw_adapter/glfw_adapter_main.cc
  )
  target_include_directories(quadrotor_sim_glfw_adapter PRIVATE
    ${PROJ_ROOT}/include
    ${PROJ_ROOT}/src/core
    ${GLFW_INCLUDE_DIRS}
  )
  target_compile_options(quadrotor_sim_glfw_adapter PRIVATE -Wall -Wextra)
  target_link_libraries(quadrotor_sim_glfw_adapter PRIVATE
    mujoco
    ${GLFW_LIBRARIES}
    pthread
    rt
    ${CMAKE_DL_LIBS}
  )
endif()
